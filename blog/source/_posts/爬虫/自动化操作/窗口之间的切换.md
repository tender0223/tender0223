 ```python
 from selenium.webdriver import Edge
 from selenium.webdriver.common.keys import Keys
 import time
 web = Edge()
 web.get("http://www.lagou.com")
 print(web.title)
 
 el = web.find_element_by_xpath('//*[@id="changeCityBox"]/ul/li[1]/a')
 el.click()
 time.sleep(2)
 
 inputpage = web.find_element_by_xpath('//*[@id="search_input"]').send_keys('python',Keys.ENTER)
 time.sleep(1)
 
 #点击之后 一定会进入新窗口
 newPgae = web.find_element_by_xpath('//*[@id="openWinPostion"]').click()
 time.sleep(2)
 
 #在selenium 的眼中 当前窗口仍然是之前的 而不是现在最新的窗口
 web.switch_to.window(web.window_handles[-1])  #选择最新的那个窗口
 eee = web.find_element_by_xpath('//*[@id="job_detail"]/dd[2]/div').text
 print(eee)
 
 time.sleep(2)
 
 #关闭窗口
 web.close()
 # 窗口关闭之后 让selenium视角转回到第一个窗口
 web.switch_to.window(web.window_handles[0])
 print(web.find_element_by_xpath('//*[@id="openWinPostion"]').text)
 
 # 如果处理iframe 必须要先拿到iframe  然后切换到iframe 视角 然后才可以拿到数据
 ```
