# 练习

```python
import re
# findall:匹配字符串中所有的符合正则的内容
list = re.findall('\d+','我的电话号是：10086，tom的电话是10010')
print('list的',list)

# finditer： 匹配字符串中所有的内容[返回的是迭代器]，从迭代器中拿到内容需要.group（）
iter = re.finditer('\d+','我的电话号是：10086，tom的电话是10010')
# print('iter的',iter)  #<callable_iterator object at 0x000001A51C410D00>
# callable_iterator 是一个迭代器
for i in iter:
    # print(i) #<re.Match object; span=(7, 12), match='10086'>
    print('iter的',i.group())
search = re.search('\d+','我的电话号是：10086，tom的电话是10010')
# 找到一个结果就会返回 返回的数据是match数据 拿数据需要.group()
print(search) # <re.Match object; span=(7, 12), match='10086'>
print('search的',search.group())

match = re.match('\d+','10086，tom的电话是10010')
print(match)

#预加载正则表达式
obj = re.compile(r'\d+')
list = obj.finditer('我的电话号是：10086，tom的电话是10010')
for it in list:
    print('it',it.group())
    
s = """
<div class='西游记'><span id='10010'>中国联通</span></div>
<div class='123'><span id='10010'>12333</span></div>
<div class='333'><span id='10010'>444</span></div>
<div class='444'><span id='10010'>555</span></div>
"""
obj = re.compile(r"<div class='(?P<class>.*?)'><span id='(?P<id>\d+)'>(?P<hh>.*?)</span></div>", re.S)
list = obj.finditer(s)
for i in list:
    print('obj-class',i.group('class'))
    print('obj-id',i.group('id'))
    print('obj-hh',i.group('hh'))

```
