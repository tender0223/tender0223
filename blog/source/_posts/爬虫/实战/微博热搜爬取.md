## 阅读前提

我在写这个脚本的代码的时候 我开始慢慢用模块化写作了 不然一次维修的地方需要用的太多了 只有用了模块化 才会让我更加舒服 （啊 真舒服啊），

本文章底部 会有整个代码的包 包括exe文件 需要的自取即可

在使用脚本时候 如果不关闭的话 将会每隔上 1小时+0.5秒的时间 会有一次对微博热搜排行榜的获取

```python
# coding=utf-8  
# 这个是对整个代码的编码器进行统一的设定
import requests
import json
import time
# 引入相关包
url = 'https://weibo.com/ajax/side/hotSearch'
header = {
    "User-Agent":"Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/117.0.0.0 Safari/537.36 Edg/117.0.2045.47"
}
# 发送请求
respond = requests.get(url, headers=header)
print(respond) # 看一看 是否发包成功
respondData = respond.text
respondData = json.loads(respondData) # 将dict格式的响应数据 通过json包 来转化成str类型的 才可以进行后续处理
t = time.localtime()  #初始化时间包 方便后面一小时进行一次热度搜查
localTime = str(t.tm_year) +'_'+ str(t.tm_mon) +'_'+ str(t.tm_mday) +'_'+ str(t.tm_hour) 
def getHot():  #定义了一个函数 获取热度榜的函数
    for i in respondData['data']['realtime'][0:6]:  #获取第一个到第六个之间的热度 
        category = i['category']
        note = i['note']
        realpos = i['realpos']
        label_name = i['label_name']
        str = f'类型: {category},标题: {note},排名: {realpos},icon标签: {label_name}'
        download(str)  #使用函数download函数进行下载

def getHotgov(): #定义了一个新函数 获取热搜榜置顶的内容
    note = respondData['data']['hotgov']['note']
    icon = respondData['data']['hotgov']['icon_desc']
    HotgovUrl = respondData['data']['hotgov']['url']
    str = f'类型: 置顶,标题: {note},icon标签: {icon},链接: {HotgovUrl}' #同时这边我再次获取了 这个内容的url 没有给上面的那些加 是因为那些好像都是讨论组 我不太会玩wb 所以就每加
    download(str)
def download(list): # 定义了下载函数 用来下载内容 
    with open (f'./{localTime}.txt',mode='a')as f: #文件名字就叫 “localTime” 这个time是由 初始化时间包下一行定义的 根据当前时间进行设定txt文本名称 模式是 a 代表着 如果没有文件则创建 如果有则持续写入不顶替原内容 
        f.write(f'{list}\n') # 每写入一行则会自动换行
if __name__ == '__main__':
    while(True): # 永远执行
        getHot()
        print('all getHot is over ~@-v-@')
        getHotgov()
        print('all getHotgov is over ~@-v-@')
        time.sleep(3600.5) #执行一次休息 1h+0.5s

```
